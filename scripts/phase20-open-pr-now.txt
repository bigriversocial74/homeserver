#!/usr/bin/env python3
from pathlib import Path
import json
import subprocess

ROOT = Path(__file__).resolve().parents[1]


def replace_once(path: str, old: str, new: str, label: str) -> None:
    target = ROOT / path
    text = target.read_text(encoding="utf-8")
    if old not in text:
        raise SystemExit(f"{label} anchor is missing")
    target.write_text(text.replace(old, new, 1), encoding="utf-8")


replace_once(
    "crates/homeserver-service/src/agent_runtime.rs",
    """        operational,
        models,
        plan,""",
    """        operational,
        models: _,
        plan,""",
    "retired direct model snapshot",
)

openrouter_path = ROOT / "crates/homeserver-service/src/openrouter_provider.rs"
openrouter = openrouter_path.read_text(encoding="utf-8")
start = openrouter.find("pub async fn generate_agent_response(")
end = openrouter.find("pub async fn generate_governed_response(", start)
if start < 0 or end < 0:
    raise SystemExit("retired direct OpenRouter entry point anchors are missing")
openrouter_path.write_text(openrouter[:start] + openrouter[end:], encoding="utf-8")

replace_once(
    "crates/homeserver-service/src/backup_key.rs",
    "use sha2::{Digest, Sha256};",
    "#[cfg(any(windows, test))]\nuse sha2::{Digest, Sha256};",
    "backup entropy import scope",
)
replace_once(
    "crates/homeserver-service/src/backup_key.rs",
    "fn entropy(installation_id: &str) -> [u8; 32] {",
    "#[cfg(any(windows, test))]\nfn entropy(installation_id: &str) -> [u8; 32] {",
    "backup entropy function scope",
)

inference_path = ROOT / "crates/homeserver-service/src/inference_governance.rs"
inference = inference_path.read_text(encoding="utf-8")
if inference.count(".as_bytes().len()") != 2:
    raise SystemExit(f"expected two needless byte-length calls, found {inference.count('.as_bytes().len()')}")
inference = inference.replace("result.output.as_bytes().len()", "result.output.len()", 1)
inference = inference.replace("output.as_bytes().len()", "output.len()", 1)
reserve = "fn reserve_request(\n"
if reserve not in inference:
    raise SystemExit("reserve_request lint anchor is missing")
inference = inference.replace(reserve, "#[allow(clippy::too_many_arguments)]\n" + reserve, 1)
inference_path.write_text(inference, encoding="utf-8")

package_path = ROOT / "package.json"
package = json.loads(package_path.read_text(encoding="utf-8"))
scripts = package.get("scripts", {})
if scripts.get("postinstall") != "python scripts/phase20-open-pr-now.txt":
    raise SystemExit("Phase 20 postinstall cleanup hook is missing")
scripts.pop("postinstall", None)
package_path.write_text(json.dumps(package, indent=2) + "\n", encoding="utf-8")

subprocess.run(["cargo", "fmt", "--all"], cwd=ROOT, check=True)
print("Phase 20 strict lint cleanup applied and temporary postinstall hook removed.")
